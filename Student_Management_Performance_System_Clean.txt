7. AGGREGATION: STUDENT COUNT AND AVERAGE MARKS

db.students.aggregate([
  {
    $group: {
      _id: "$course",
      total_students: { $sum: 1 },
      average_marks: { $avg: "$marks" }
    }
  },
  {
    $project: {
      _id: 0,
      course: "$_id",
      total_students: 1,
      average_marks: { $round: ["$average_marks", 2] }
    }
  },
  {
    $sort: { course: 1 }
  }
])
